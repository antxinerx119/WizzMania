cmake_minimum_required(VERSION 3.16)

project(Mess_Tests VERSION 0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

enable_testing()

# Google Test Framework - Automatically fetch from GitHub
include(FetchContent)
FetchContent_Declare(
    GoogleTest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG v1.14.0
)

set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(GoogleTest)

# Common protocol and message classes (WITHOUT Qt)
add_library(MessCommon STATIC
    src/common/Protocol.cpp
    src/common/Message.cpp
)
target_include_directories(MessCommon PUBLIC src/common)
# Note: These files use Qt (QDataStream), so tests will need Qt too

# Test 1: Message Serialization
add_executable(TestMessage tests/test_message.cpp)
target_link_libraries(TestMessage PRIVATE MessCommon GTest::gtest_main)
add_test(NAME MessageTests COMMAND TestMessage)

# Test 2: Server Logic
add_executable(TestServer tests/test_server.cpp)
target_link_libraries(TestServer PRIVATE MessCommon GTest::gtest_main)
add_test(NAME ServerTests COMMAND TestServer)

# Test 3: Client Network
add_executable(TestClientNetwork tests/test_clientnetwork.cpp)
target_link_libraries(TestClientNetwork PRIVATE MessCommon GTest::gtest_main)
add_test(NAME ClientNetworkTests COMMAND TestClientNetwork)

# Make test targets easily accessible
add_custom_target(run_tests
    COMMAND ctest --verbose
    DEPENDS TestMessage TestServer TestClientNetwork
)

message(STATUS "Tests configured (note: still requires Qt6 at runtime)")
